#include <Wire.h>
#include <Adafruit_BME280.h>

// --- SENSOR OBJECTS ---
Adafruit_BME280 bme;

// --- PIN ASSIGNMENTS ---
const int ANEMOMETER_PIN = A0; // Brown wire goes here (Analog in). Blue goes to Ground.
const int RAIN_PIN = 7;        // RG-11 relay output

void setup() {
  Serial.begin(9600);
  
  // Set rain pin with an internal pullup 
  // (Assuming the RG-11 relay pulls the circuit to ground when wet)
  pinMode(RAIN_PIN, INPUT_PULLUP); 

  // Initialize I2C Sensor
  // Make sure your BME280 is wired to A4 (SDA) and A5 (SCL)
  bme.begin(0x76); 
}

void loop() {
  // 1. Read Ambient Data (BME280)
  float amb_c = bme.readTemperature();
  float amb_f = (amb_c * 9.0 / 5.0) + 32.0;
  float hum = bme.readHumidity();
  float pres = bme.readPressure() / 100.0F;

  // 2. Read & Map Anemometer (0-3.8V = 0-70 m/s)
  int raw_adc = analogRead(ANEMOMETER_PIN);
  
  // The Nano ADC is 10-bit (0-1023) running on a 5V reference.
  float voltage = (raw_adc / 1023.0) * 5.0;
  
  // Clamp the voltage at 3.8V max to prevent math errors if spikes occur
  if (voltage > 3.8) voltage = 3.8;
  if (voltage < 0.0) voltage = 0.0;

  // Map the voltage to meters per second, then convert to MPH
  float wind_ms = (voltage / 3.8) * 70.0;
  float wind_mph = wind_ms * 2.23694;

  // 3. Read RG-11 Rain Sensor 
  // (Assuming LOW = Wet based on the pullup resistor)
  int rain_state = digitalRead(RAIN_PIN);
  int is_wet = (rain_state == LOW) ? 1 : 0;

  // 4. Build and Send the Data Payload
  // Using println() on the last item to send the standard \r\n newline characters
  Serial.print("WEATHER:");
  Serial.print("WIND:"); Serial.print(wind_mph, 2);
  Serial.print(",AMB:"); Serial.print(amb_f, 2);
  Serial.print(",HUM:"); Serial.print(hum, 2);
  Serial.print(",PRES:"); Serial.print(pres, 2);
  Serial.print(",RAIN:"); Serial.println(is_wet); 

  // Wait 1 seconds before sending the next batch
  delay(1000); 
}
